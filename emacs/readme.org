* Overview
:PROPERTIES:
:CUSTOM_ID: h-e5db349d
:END:

A literate, distribution-independent configuration for Emacs 31+.
It works the same whether you start plain =emacs= or run =emacs --daemon= and connect with =emacsclient -c=.

Tangled files:
- =early-init.el= – GC, native-comp and GUI suppression (runs before package.el)
- =init.el= – everything else, in the order of the sections below

Conventions:
- Every =use-package= block either has a trigger (=:hook=, =:bind=, =:commands=, =:mode=) or is explicitly =:demand t=.
  Nothing relies on =use-package-always-defer=.
- Built-in features use =:ensure nil=.  Packages that are not on an ELPA use =:vc= (built into Emacs 30+).
- Extra Elisp dropped into =load.d/= next to =init.el= is loaded at the end of startup.

  ;;; early-init.el --- Early init -*- lexical-binding: t; no-byte-compile: t -*-
  ;; Generated from readme.org – edit the Org file, not this one.

  ;; GC: push the threshold up for startup.  It is restored on
  ;; `emacs-startup-hook', which also runs under --daemon.
  (setq gc-cons-threshold  most-positive-fixnum
        gc-cons-percentage 0.6)

  ;; An empty `file-name-handler-alist' speeds up every load during startup.
  (defvar my--file-name-handler-alist file-name-handler-alist)
  (setq file-name-handler-alist nil)

  (add-hook 'emacs-startup-hook
            (lambda ()
              (setq file-name-handler-alist
                    (delete-dups (append file-name-handler-alist
                                         my--file-name-handler-alist))
                    gc-cons-threshold  (* 16 1024 1024)
                    gc-cons-percentage 0.1))
            90)

  ;; Native compilation: keep JIT enabled, but do not pop up warning buffers.
  (setq native-comp-async-report-warnings-errors 'silent)
  (when (and (fboundp 'startup-redirect-eln-cache)
             (boundp 'native-comp-eln-load-path))
    (startup-redirect-eln-cache
     (expand-file-name "eln-cache/" user-emacs-directory)))

  ;; package.el is initialised by hand in init.el.
  (setq package-enable-at-startup nil)

  ;; Enable package quickstart - creates a cache that skips package checks
  (setq package-quickstart t)

  ;; No flash of an unstyled frame.
  (push '(menu-bar-lines . 0)          default-frame-alist)
  (push '(tool-bar-lines . 0)          default-frame-alist)
  (push '(vertical-scroll-bars . nil)  default-frame-alist)
  (setq frame-inhibit-implied-resize t
        use-file-dialog              nil
        inhibit-startup-screen       t
        inhibit-startup-message      t
        inhibit-startup-echo-area-message user-login-name)

** Package archives and use-package
:PROPERTIES:
:CUSTOM_ID: h-f2eb6f8e
:END:

ELPA archives win over MELPA when a package exists in both.
The ensure function installs a package only if it is not already installed
or loadable (so system or Guix provided packages are respected), and refreshes
the archive index the first time something is missing.

#+BEGIN_SRC emacs-lisp
(setq package-archives
      '(("gnu"    . "https://elpa.gnu.org/packages/")
        ("nongnu" . "https://elpa.nongnu.org/nongnu/")
        ("melpa"  . "https://melpa.org/packages/"))
      package-archive-priorities
      '(("gnu" . 3) ("nongnu" . 2) ("melpa" . 1)))

(package-initialize)

(defun my/use-package-ensure (name ensure _state)
  "Install the packages named by ENSURE (the parsed :ensure arguments).
An argument of t means NAME itself; nil means nothing to install."
  (dolist (arg ensure)
    (let ((pkg (if (eq arg t) name arg)))
      (when (and pkg
                 (not (package-installed-p pkg))
                 (not (locate-library (symbol-name pkg))))
        (unless (assq pkg package-archive-contents)
          (package-refresh-contents))
        (package-install pkg)))))

(setq use-package-ensure-function #'my/use-package-ensure
      use-package-always-ensure   nil  ;; do not wanna load 50+ package.
      use-package-enable-imenu-support t
      use-package-expand-minimally t)
(require 'use-package)
#+END_SRC

(defun my/git-config (key)
  "Return git config KEY, or nil when git or the key is missing."
  (when (executable-find "git")
    (let ((out (with-temp-buffer
                 (when (zerop (call-process "git" nil t nil "config" "--get" key))
                   (string-trim (buffer-string))))))
      (and out (not (string-empty-p out)) out))))

(setq user-full-name    (or (my/git-config "user.name")  (getenv "USER")  "user")
      user-mail-address (or (my/git-config "user.email") (getenv "EMAIL") "user@example.com")
      org-directory     (expand-file-name "~/Documents/org/"))

;; Extra binary locations (Guix, user installs, system).  Appended, so your
;; own PATH order wins.  Both `exec-path' and the PATH variable are updated.
(dolist (dir '("~/.local/bin"
               "~/.guix-profile/bin"
               "~/.config/guix/current/bin"
               "/run/current-system/profile/bin"
               "/usr/local/bin"
               "/usr/bin"))
  (let ((dir (expand-file-name dir)))
    (when (and (file-directory-p dir) (not (member dir exec-path)))
      (setq exec-path (append exec-path (list dir)))
      (setenv "PATH" (concat (getenv "PATH") path-separator dir)))))

    ;; ── 2. Language version (Uses project venv, cached) ──
    (defvar-local my/lang-version-cache nil
      "Cache for the modeline language version so it only runs once per buffer.")

    (defun my/modeline-lang-version ()
      "Return language name + version, checking project venvs first."
      (or my/lang-version-cache
          (setq my/lang-version-cache
                (cond
                 ;; PYTHON: Use python-shell-interpreter (set by uv/venv activator)
                 ((derived-mode-p 'python-mode 'python-ts-mode 'python-base-mode)
                  (condition-case nil
                      (let* ((py-exe (or (bound-and-true-p python-shell-interpreter) 
                                         "python3"))
                             (out (and (file-executable-p py-exe)
                                       (shell-command-to-string 
                                        (format "%s --version 2>&1" py-exe))))
                             (ver (and out
                                       (string-match "Python \\([0-9.]+\\)" out)
                                       (match-string 1 out))))
                        (if ver (concat "Py " ver) "Py ?"))
                    (error "Py ?")))
                 
                 ;; RUST
                 ((derived-mode-p 'rustic-mode 'rust-ts-mode 'rust-mode)
                  (condition-case nil
                      (let ((out (shell-command-to-string "rustc --version 2>&1")))
                        (if (string-match "rustc \\([0-9.]+\\)" out)
                            (concat "Rust " (match-string 1 out))
                          "Rust"))
                    (error "Rust")))
                    
                 ;; GO
                 ((derived-mode-p 'go-mode 'go-ts-mode)
                  (condition-case nil
                      (let ((out (shell-command-to-string "go version 2>&1")))
                        (if (string-match "go\\([0-9.]+\\)" out)
                            (concat "Go " (match-string 1 out))
                          "Go"))
                    (error "Go")))
                    
                 ;; FALLBACK
                 (t (format-mode-line mode-name))))))

    ;; ── 3. Git branch helper (robust parsing) ──
    (defvar-local my/branch-cache nil
      "Cache for git branch to avoid re-parsing on every redraw.")

    (defun my/modeline-branch ()
      "Return just the branch name from vc-mode."
      (or my/branch-cache
          (setq my/branch-cache
                (when vc-mode
                  (let* ((raw (string-trim-left vc-mode))
                         (after-colon (if (string-match ":" raw)
                                          (substring raw (match-end 0))
                                        raw)))
                    (string-trim-right after-colon "\\*"))))))

      "Return the full file path (abbreviated with ~)."
      (if buffer-file-name
          (abbreviate-file-name buffer-file-name)
        (buffer-name)))

  ;; ── 5. Verbose Flymake diagnostic counter (FIXED) ──
  (defun my/modeline-flymake ()
    "Return verbose Flymake diagnostic counts using public API."
    (when (bound-and-true-p flymake-mode)
      (condition-case nil
          (let* (;; Try public API first, then internal variable
                 (diags (cond
                         ((fboundp 'flymake-diagnostics)
                          (flymake-diagnostics))
                         ((boundp 'flymake--diagnostics)
                          flymake--diagnostics)
                         (t nil)))
                 (err 0) (warn 0) (note 0))
            (when (listp diags)
              (dolist (diag diags)
                (let ((type (condition-case nil
                                (flymake-diagnostic-type diag)
                              (error
                               ;; Fallback: diagnostics are vectors, type at index 2
                               (and (vectorp diag) (aref diag 2))))))
                  (pcase type
                    (:error   (cl-incf err))
                    (:warning (cl-incf warn))
                    (:note    (cl-incf note))
                    (_        (cl-incf note))))))
            (if (or (> err 0) (> warn 0) (> note 0))
                (format "E:%d W:%d N:%d" err warn note)
              "✓"))
        (error "✓"))))

    ;; ── 6. DYNAMIC width calculator for the right side ──
    (defun my/modeline-right-reserve ()
      "Calculate the exact character width of the right-side content."
      (let* ((flymake-str (or (my/modeline-flymake) ""))
             (lang-str (my/modeline-lang-version))
